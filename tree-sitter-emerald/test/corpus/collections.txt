h: Hash[Int64, Int64] = {1 => 10, 2 => 20}
puts h[1]

other: Boolean = false
if flag do
  puts 1
end
if other do
  puts 2
else
  puts 3
end

s: String? = nil
if s == nil do
  puts 4
end
